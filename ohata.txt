#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

typedef struct livro {
    int reg;            // registro do livro
    char titulo[80];    // título do livro
    char status;        // [L]ivre ou [E]mprestado
} livro;

typedef struct aluno {
    char RA[7];         // RA do aluno
    char nome[80];      // nome do aluno
    int numreg;         // número do registro do livro
    int dias;           // tempo de permanência - em dias
} aluno;

void aloca_livro(livro **pliv, int qliv);
void aloca_aluno(aluno **pal, int qal);
void cadastra_livro(livro *pliv, int qliv);
void cadastra_aluno(aluno *pal, livro *pliv, int qliv);
void devolucao(aluno *pal, int qal, livro *pliv, int qliv);
int busca_livro(livro *pliv, int qliv, char *titulo);
int busca_RA(aluno *pal, int qal, char *aux);
void mostra_livro(livro *pliv, int qliv);
void mostra_aluno(aluno *pal, int qal);

int main() {
    livro *liv = NULL;
    aluno *alu = NULL;
    int op, cont = 0;

    aloca_livro(&liv, 5);
    cadastra_livro(liv, 5);

    do {
        system("cls"); // ou system("clear") no Linux
        printf("\n[1] Emprestimo");
        printf("\n[2] Devolucao");
        printf("\n[3] Mostrar Alunos");
        printf("\n[4] Fim");
        printf("\nOpcao: ");
        scanf("%d", &op);
        fflush(stdin);

        switch (op) {
        case 1:
            mostra_livro(liv, 5);
            aloca_aluno(&alu, cont + 1);
            cadastra_aluno(alu + cont, liv, 5);
            cont++;
            break;
        case 2:
            devolucao(alu, cont, liv, 5);
            break;
        case 3:
            mostra_aluno(alu, cont);
            break;
        }
    } while (op != 4);

    free(liv);
    free(alu);
    return 0;
}

//-----------------------------------------------------

void aloca_livro(livro **pliv, int qliv) {
    if ((*pliv = (livro*) realloc(*pliv, qliv * sizeof(livro))) == NULL)
        exit(1);
}

void aloca_aluno(aluno **pal, int qal) {
    if ((*pal = (aluno*) realloc(*pal, qal * sizeof(aluno))) == NULL)
        exit(1);
}

void cadastra_livro(livro *pliv, int qliv) {
    livro *p;
    int reg = 1;

    for (p = pliv; p < pliv + qliv; p++) {
        p->reg = reg++;
        printf("Titulo do livro %d: ", p->reg);
        fgets(p->titulo, 80, stdin);
        p->titulo[strcspn(p->titulo, "\n")] = 0;
        p->status = 'L';
    }
}

void cadastra_aluno(aluno *pal, livro *pliv, int qliv) {
    char titulo[80];
    int numliv;

    printf("\nTitulo do livro desejado: ");
    fgets(titulo, 80, stdin);
    titulo[strcspn(titulo, "\n")] = 0;

    numliv = busca_livro(pliv, qliv, titulo);

    if (numliv == -1)
        printf("\nNao ha livros disponiveis com esse titulo.\n\n");
    else {
        pal->numreg = numliv;
        printf("\nRA do aluno: ");
        gets(pal->RA);
        fflush(stdin);
        printf("\nNome: ");
        gets(pal->nome);
        fflush(stdin);
        printf("\nQuantos dias: ");
        scanf("%d", &pal->dias);
        fflush(stdin);
        printf("\nEmprestimo realizado com sucesso!\nLivro: %d\n\n", pal->numreg);
    }
    system("pause");
}

void devolucao(aluno *pal, int qal, livro *pliv, int qliv) {
    char aux_RA[7];
    int pos, aux_reg, tempo;
    char titulo[80];

    printf("\nRA do aluno para devolucao: ");
    gets(aux_RA);
    fflush(stdin);

    pos = busca_RA(pal, qal, aux_RA);

    if (pos == -1)
        printf("\nRA invalido.\n");
    else {
        aux_reg = (pal + pos)->numreg;
        tempo = (pal + pos)->dias;
        (pliv + aux_reg - 1)->status = 'L';

        strcpy(titulo, (pliv + aux_reg - 1)->titulo);
        printf("\nNome: %s\nTitulo: %s\nDias: %d\n", (pal + pos)->nome, titulo, tempo);
        printf("\nDevolucao realizada com sucesso!\n\n");
    }
    system("pause");
}

int busca_livro(livro *pliv, int qliv, char *titulo) {
    livro *p;
    for (p = pliv; p < pliv + qliv; p++) {
        if (strcasecmp(p->titulo, titulo) == 0 && p->status == 'L') {
            p->status = 'E';
            return p->reg;
        }
    }
    return -1;
}

int busca_RA(aluno *pal, int qal, char *aux) {
    aluno *p;
    for (p = pal; p < pal + qal; p++) {
        if (strcmp(p->RA, aux) == 0)
            return (p - pal);
    }
    return -1;
}

void mostra_livro(livro *pliv, int qliv) {
    livro *p;
    for (p = pliv; p < pliv + qliv; p++)
        printf("\nRegistro: %d\nTitulo: %s\nStatus: %c\n", p->reg, p->titulo, p->status);
    printf("\n\n");
    system("pause");
}

void mostra_aluno(aluno *pal, int qal) {
    aluno *p;
    for (p = pal; p < pal + qal; p++)
        printf("\nRA: %s\nNome: %s\nLivro: %d\nDias: %d\n", p->RA, p->nome, p->numreg, p->dias);
    printf("\n\n");
    system("pause");
}
